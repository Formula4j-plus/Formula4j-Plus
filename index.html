快速开始

1. 添加依赖

<!-- Maven --> <dependency> <groupId>com.formula</groupId> <artifactId>formula4j-plus</artifactId> <version>2.2.0</version> </dependency>

2. 基本使用

import com.formula.calculator.FormulaCalculator; import com.alibaba.fastjson.JSONObject; // 准备数据 JSONObject data = new JSONObject(); data.put("price", 100); data.put("quantity", 5); // 计算公式 String formula = "${price} * ${quantity}"; Object result = FormulaCalculator.calculate(formula, data); System.out.println("结果: " + result); // 输出: 500.0

3. 使用函数

// 使用SUM函数 String formula = "SUM(10, 20, 30)"; Object result = FormulaCalculator.calculate(formula, data); // 结果: 60.0 // 使用IF函数 String formula2 = "IF(10 > 5, 100, 0)"; Object result2 = FormulaCalculator.calculate(formula2, data); // 结果: 100.0

使用示例

使用 FormulaData 结构

import com.formula.calculator.model.FormulaData; import com.formula.calculator.model.FieldMark; // 构建FormulaData FormulaData formulaData = new FormulaData(); formulaData.setTxt("SUM(单价,数量) * (1 - 折扣率)"); // 添加字段映射 List<FieldMark> marks = new ArrayList<>(); marks.add(new FieldMark("price", "单价")); marks.add(new FieldMark("quantity", "数量")); marks.add(new FieldMark("discount", "折扣率")); formulaData.setMarks(marks); // 准备数据 JSONObject data = new JSONObject(); data.put("price", 100); data.put("quantity", 5); data.put("discount", 0.1); // 计算 Object result = FormulaCalculator.calculate(formulaData, data); // 结果: 450.0

自定义函数

// 注册Java Lambda函数 FormulaCalculator.registerCustomFunction("DOUBLE", (params, data, mapping) -> { double value = Double.parseDouble(params.trim()); return String.valueOf(value * 2); }); // 使用 String formula = "DOUBLE(10)"; Object result = FormulaCalculator.calculate(formula, data); // 结果: 20.0

一次性加载 Formula.js 库

// 加载Formula.js库 FormulaCalculator.loadFormulaJsLibrary("file:///D:/formula.js"); // 加载后，所有函数都可以直接使用 String formula = "SUM(1, 2, 3) + ABS(-5) + SQRT(16)"; Object result = FormulaCalculator.calculate(formula, data); // 结果: 15.0
